#include <iostream>
#include <string>
using namespace std;

int main() {
    // Установка кодировки для корректного отображения кириллицы в консоли (Windows)
    std::setlocale(LC_ALL, "Russian");

    int choice = 0;

    cout << "=== Простая игра: Правда или Неправда ===\n";
    cout << "Вопрос: Земля имеет форму идеального шара?\n";
    cout << "1. Правда\n";
    cout << "2. Неправда\n";
    cout << "Ваш ответ (1 или 2): ";
    
    cin >> choice;

    // Проверка ответа с использованием конструкции if-else
    if (choice == 1) {
        cout << "\nНеверно! Земля сплюснута у полюсов (это геоид), а не идеальный шар.\n";
    } 
    else if (choice == 2) {
        cout << "\nВерно! Земля не является идеальным шаром.\n";
    } 
    else {
        cout << "\nОшибка: вы выбрали вариант, которого нет в списке.\n";
    }

    return 0; //;:AALALALLALAALALAL eqrqwr
}
