#include <iostream>
#include <string>
#include <vector>

using namespace std;

// Структура для опису вікон системи
struct Screen {
    int id;
    string title;
    string description;
};

void showMenu() {
    cout << "=========================================" << endl;
    cout << "   СИСТЕМА УПРАВЛІННЯ ІНТЕРФЕЙСОМ (ОПІ)   " << endl;
    cout << "=========================================" << endl;
    cout << "1. Вікно авторизації (Auth)" << endl;
    cout << "2. Головна панель (Dashboard)" << endl;
    cout << "3. Каталог об'єктів (Table/List)" << endl;
    cout << "4. Форма створення/редагування (Form)" << endl;
    cout << "5. Профіль користувача (Settings)" << endl;
    cout << "0. Вихід" << endl;
    cout << "=========================================" << endl;
}

int main() {
    system("chcp 65001 > nul"); // Підтримка української мови в консолі

    vector<Screen> screens = {
        {1, "Авторизація", "Форма входу з полями Email/Password та кнопкою Увійти"},
        {2, "Головна панель", "Дашборд зі статистикою, графіками та швидкими діями"},
        {3, "Каталог об'єктів", "Таблиця даних з можливістю фільтрації та пошуку"},
        {4, "Форма створення", "Форма введення нових даних з валідацією полів"},
        {5, "Профіль та налаштування", "Вікно керування особистими даними користувача"}
    };

    int choice = -1;

    while (choice != 0) {
        showMenu();
        cout << "Оберіть вікно для перегляду детальної інформації (0-5): ";
        cin >> choice;

        if (choice >= 1 && choice <= 5) {
            cout << "\n-----------------------------------------" << endl;
            cout << "Модуль №" << screens[choice - 1].id << ": " << screens[choice - 1].title << endl;
            cout << "Опис: " << screens[choice - 1].description << endl;
            cout << "-----------------------------------------\n" << endl;
        } else if (choice != 0) {
            cout << "\n[Помилка] Некоректний вибір! Спробуйте ще раз.\n" << endl;
        }
    }

    cout << "Програму завершено. Дякуємо!" << endl;
    return 0;
}
