#include <iostream>

using namespace std;

int main() {
    // Встановлюємо українську локаль у консолі (для коректного виведення тексту)
    setlocale(LC_ALL, "Ukrainian");

    double a, b, x;
    double result;

    cout << "=== Програма для обчислення рівняння: (a) + (b) - (x)^2 ===" << endl;

    // Введення даних користувачем
    cout << "Введіть перше число (a): ";
    cin >> a;

    cout << "Введіть друге число (b): ";

    cout << "Введіть число для квадрата (x): ";
    cin >> x;

    // Обчислення: a + b - x^2
    result = a + b - sqrt(x);

    // Виведення результату
    cout << "\n--- Результат ---" << endl;
    cout << "Обчислення: " << a << " + " << b << " - (" << x << ")^2" << endl;
    cout << "Кінцевий результат: " << result << endl;

    return 0;
}
